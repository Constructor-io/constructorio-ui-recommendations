# useRecommendationResults

A hook that fetches and manages recommendation results from the Constructor.io Recommendations API. Use this hook when you need full control over how recommendations are rendered while letting the library handle data fetching and state management.

## When to Use

Use this hook when you need to:
- Build a completely custom recommendations UI
- Integrate recommendations into an existing component structure
- Access the raw recommendation data for custom processing
- Implement custom loading and error states

**Example use cases:**
- Building a custom grid layout instead of the default carousel
- Integrating recommendations into a sidebar widget
- Creating a recommendations page with filtering and sorting
- Implementing server-side rendering with initial data

## Usage

This hook must be used within a `CioRecommendationProvider` context.

```jsx
import {
  CioRecommendationProvider,
  useCioRecommendationContext,
  useRecommendationResults
} from '@constructor-io/constructorio-ui-recommendations';

function CustomRecommendations() {
  // Access context values
  const { podId } = useCioRecommendationContext();

  // Fetch recommendations
  const { status, data, refetch } = useRecommendationResults();

  // Render your custom UI
  return (
    <div>
      <h2>{data.response.pod.displayName}</h2>
      {data.response.results.map((item) => (
        <div key={item.id}>{item.name}</div>
      ))}
    </div>
  );
}

function App() {
  return (
    <CioRecommendationProvider apiKey="YOUR_API_KEY" podId="YOUR_POD_ID">
      <CustomRecommendations />
    </CioRecommendationProvider>
  );
}
```

## Props

| Property | Type | Description |
| -------- | ---- | ----------- |
| `initialRecommendationResponse` | `ApiRecommendationsResponse` | Initial value for recommendation results. Useful for Server-Side Rendering (SSR) to hydrate the component on the client |

---

## Return Value

| property | type                            | description                                                                                   |
| -------- | ------------------------------- | --------------------------------------------------------------------------------------------- |
| status   | `RequestStatus`                 | The current status of the recommendation request (eg. `IDLE`, `FETCHING`, `SUCCESS`, `ERROR`) |
| message  | `string` \| `null`              | Any error message encountered during the request                                              |
| data     | `RecommendationsData` \| `null` | The transformed recommendation data                                                           |
| refetch  | () => void                      | A function to manually refetch the recommendation results                                     |


### Type Guards

The hook also exports type guard functions for narrowing the return type:

| Function          | Description                                            |
| ----------------- | ------------------------------------------------------ |
| `isResponseLoaded`  | Returns `true` when `status` is `SUCCESS`              |
| `isResponseLoading` | Returns `true` when `status` is `FETCHING` or `IDLE`   |
| `isResponseError`   | Returns `true` when `status` is `ERROR`                |

```jsx
import {
  useRecommendationResults,
  isResponseLoaded,
  isResponseLoading,
  isResponseError
} from '@constructor-io/constructorio-ui-recommendations';

function MyComponent() {
  const response = useRecommendationResults();

  if (isResponseLoading(response)) {
    return <LoadingSpinner />;
  }

  if (isResponseError(response)) {
    return <ErrorMessage message={response.message} />;
  }

  if (isResponseLoaded(response)) {
    // TypeScript knows response.data is defined here
    return <RecommendationsList items={response.data.response.results} />;
  }
}
```

## Interactive Example

<Canvas of={UseRecommendationResultsStories.Default} />

```jsx
import {
  CioRecommendationProvider,
  useCioRecommendationContext,
  useRecommendationResults
} from '@constructor-io/constructorio-ui-recommendations';

function CustomRecommendations() {
  const { podId } = useCioRecommendationContext();
  const { status, data, message, refetch } = useRecommendationResults();

  if (status === 'fetching' || status === 'idle') {
    return (
      <div className='hook-example-container'>
        <div className='hook-example-loading'>Loading recommendations...</div>
      </div>
    );
  }

  if (status === 'error') {
    return (
      <div className='hook-example-container'>
        <div className='hook-example-error'>Error: {message}</div>
      </div>
    );
  }

  return (
    <div className='hook-example-container'>
      <div className='hook-example-header'>
        <div>
          <h2 className='hook-example-title'>{data.response.pod.displayName}</h2>
          <span className='hook-example-pod-id'>Pod: {podId}</span>
        </div>
        <button className='hook-example-refresh' type='button' onClick={refetch}>
          Refresh
        </button>
      </div>
      <div className='hook-example-grid'>
        {data.response.results.slice(0, 6).map((item) => (
          <div key={item.id} className='hook-example-card'>
            <img src={item.imageUrl} alt={item.name} />
            <h3>{item.name}</h3>
            {item.price && <span className='hook-example-price'>${item.price}</span>}
          </div>
        ))}
      </div>
    </div>
  );
}

function App() {
  return (
    <CioRecommendationProvider apiKey="YOUR_API_KEY" podId="YOUR_POD_ID">
      <CustomRecommendations />
    </CioRecommendationProvider>
  );
}
```
